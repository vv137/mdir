cmake_minimum_required(VERSION 3.25)
project(MDIRPrototypeTransplantGate LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
find_package(CUDAToolkit 13.2 REQUIRED)
add_executable(prototype-entry-format-test EntryFormatTest.cpp)
add_executable(prototype-gate-input-audit GateInputAudit.cpp)
add_executable(prototype-initial-entry-gate InitialEntryGate.cu)
target_compile_definitions(prototype-initial-entry-gate PRIVATE FUSE_MINB=1)
target_compile_options(prototype-initial-entry-gate PRIVATE
  $<$<COMPILE_LANGUAGE:CUDA>:-O3;--generate-line-info>)
target_link_libraries(prototype-initial-entry-gate PRIVATE CUDA::cudart ${CMAKE_DL_LIBS})
add_library(prototype-session SHARED PrototypeSession.cu NativeStage.cu PhysicalCoverage.cu)
target_compile_definitions(prototype-session PRIVATE FUSE_MINB=1)
target_compile_options(prototype-session PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--generate-line-info>)
set_source_files_properties(NativeStage.cu PROPERTIES COMPILE_OPTIONS "--extended-lambda")
target_link_libraries(prototype-session PRIVATE CUDA::cudart ${CMAKE_DL_LIBS})
add_library(prototype-native-adapter SHARED NativeAdapter.cu)
target_include_directories(prototype-native-adapter PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_link_libraries(prototype-native-adapter PRIVATE prototype-session CUDA::cudart ${CMAKE_DL_LIBS})
