  set(GROUPS_TEMPLATE_PYTHON ${Python_EXECUTABLE})

  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(GROUPS_TEMPLATE_PYTHON ${Python3_EXECUTABLE})
endif()

  ${CMAKE_CURRENT_BINARY_DIR}/single-pass-groups)

  ${PROJECT_SOURCE_DIR}/scripts/generate-groups-gpu-template.py)
execute_process(
  COMMAND ${GROUPS_TEMPLATE_PYTHON}
    ${PROJECT_SOURCE_DIR}/scripts/generate-groups-gpu-template.py
    ${GROUPS_SINGLE_PASS_TEMPLATE_DIR} single_pass
  RESULT_VARIABLE GROUPS_SINGLE_PASS_RESULT)
if(NOT GROUPS_SINGLE_PASS_RESULT EQUAL 0)
  message(FATAL_ERROR "Could not generate single-pass group templates")
endif()
